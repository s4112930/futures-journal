async function syncIGTrades(){ const button = document.getElementById("igSyncBtn"); const status = document.getElementById("igSyncStatus"); button.disabled = true; button.innerText = "同步中…"; try{ const endDate = new Date(); const startDate = new Date(); startDate.setFullYear( startDate.getFullYear() - 1 ); const allRaw = []; let currentStart = new Date(startDate); let batch = 0; while( currentStart < endDate ){ batch++; let currentEnd = new Date(currentStart); currentEnd.setDate( currentEnd.getDate() + 14 ); if( currentEnd > endDate ){ currentEnd = new Date(endDate); } status.innerText = "正在同步最近 1 年交易紀錄… 第 " + batch + " 段"; const url = AI_API + "/ig/transactions" + "?from=" + encodeURIComponent( toIGQueryTime(currentStart) ) + "&to=" + encodeURIComponent( toIGQueryTime(currentEnd) ); const response = await fetch(url); const data = await response.json(); if( !response.ok || !data.ok ){ throw new Error( data.error || "讀取 IG 交易失敗" ); } if( Array.isArray( data.transactions ) ){ allRaw.push( ...data.transactions ); } currentStart = new Date(currentEnd); currentStart.setSeconds( currentStart.getSeconds() + 1 ); await sleep(250); } /* 先按照 IG reference 去除 各日期區段可能重疊的資料 */ const uniqueMap = new Map(); allRaw.forEach( item => { const key = item.reference || [ item.instrumentName, item.openDateUtc, item.dateUtc, item.openLevel, item.closeLevel, item.size ].join("|"); if( !uniqueMap.has(key) ){ uniqueMap.set( key, item ); } } ); const raw = Array.from( uniqueMap.values() ); let imported = 0; let duplicated = 0; let excluded = 0; let ignored = 0; raw.forEach( item => { if( !isRealIGTrade(item) ){ ignored++; return; } const product = normalizeIGProduct( item.instrumentName ); if( !isAllowedNormalizedProduct( product ) ){ excluded++; return; } if( item.reference && trades.some( t => t.igReference === item.reference ) ){ duplicated++; return; } const converted = convertIGTrade(item); if( !converted ){ ignored++; return; } trades.push( converted ); imported++; } ); trades.sort( (a,b) => new Date( b.exitTime || b.entryTime ) - new Date( a.exitTime || a.entryTime ) ); saveTrades(); render(); status.innerText = "一年同步完成：新增 " + imported + " 筆，重複 " + duplicated + " 筆，排除其他商品 " + excluded + " 筆。共讀取 " + raw.length + " 筆 IG 紀錄。"; }catch(error){ status.innerText = "同步失敗：" + error.message; } button.disabled = false; button.innerText = "同步 IG 交易"; }
